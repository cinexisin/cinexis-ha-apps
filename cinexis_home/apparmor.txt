#include <tunables/global>

# CINEXIS Home.
#
# The narrowest profile of anything we ship, because this is the app that runs in
# every house all the time. It reads Home Assistant's API, talks outbound to
# CINEXIS, and writes one private directory.
#
# WHY THESE RULES CARRY `r` AND NOT ONLY `ix`
# -------------------------------------------
# 4.1.0-pilot.4 failed on every device with:
#
#     /bin/sh: can't open '/init': Permission denied
#
# `/init ix` grants EXECUTE. It does not grant READ. `/init` is a shell script
# (`#!/bin/sh -e`), so the kernel execs it, reads the shebang, and then launches
# `/bin/sh /init` — at which point the interpreter must READ the file it was
# handed. AppArmor refused the read, and s6 never started.
#
# The same trap applies to `/command/with-contenv`, which is also a script
# (`#!/command/execlineb -S0`). Every path here that can hold an interpreted
# script therefore carries `r` alongside `ix`. Pure ELF binaries do not need it,
# but distinguishing them per-file is a maintenance trap for no security gain:
# read on an executable the profile already permits to run grants nothing extra.
profile cinexis_home flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # No sys_admin, no net_admin, no net_raw, no net_bind_service: it listens on
  # 8099, which is unprivileged, and dials out.
  #
  # No chown/fowner either: the base image ships /etc/fix-attrs.d EMPTY, so s6
  # changes no ownership at startup. Verified against the pinned base rather than
  # assumed, because these are exactly the capabilities that get added "just in
  # case" and never removed.
  capability setgid,
  capability setuid,

  network inet  stream,
  network inet6 stream,

  # s6-svscan supervises its children and forwards shutdown.
  signal (send,receive) set=(kill,term,int,hup,cont,usr1,usr2,alrm,chld),

  # ── s6-overlay v3 ──────────────────────────────────────────────────────────
  /init                 rix,
  /command/**           rix,
  /package/**           rix,
  /etc/s6-overlay/**    rix,
  /bin/**               rix,
  /usr/bin/**           rix,

  # s6 does not merely READ its runtime tree — it WRITES executables into it at
  # startup and then runs them. stage0 generates /run/s6/basedir/bin/init and
  # execs it, which is what 4.1.0-pilot.5 was denied:
  #
  #   stage0: exec: line 87: /run/s6/basedir/bin/init: Permission denied
  #
  # `/run/** rwk` grants read, write and lock. It does not grant execute, and
  # nothing else in this profile covered these paths.
  #
  # Home Assistant's own maintained profiles list only `/run/s6/** rwix` — but
  # they also carry a blanket `file,` rule, which silently covers the rest. This
  # profile has no `file,` by design, so every tree s6 executes from is named.
  # Enumerated from a running container rather than guessed: 31 executables
  # across three roots, one of which carries a random per-boot suffix
  # (/run/s6-rc:s6-rc-init:XXXXXX). Fixing only the path in the error message
  # would have moved the failure to the next tree, not removed it.
  /run/s6/**            rwix,
  /run/service/**       rwix,
  /run/s6-rc*/**        rwix,

  # Everything else under /run stays non-executable: state, not code.
  /run/                 rwk,
  /run/**               rwk,

  # ── TLS: OpenSSL configuration and the CA trust store, read-only ───────────
  # Node aborts during startup if it cannot READ its OpenSSL config:
  #
  #   node: OpenSSL configuration error: BIO_new_file: Permission denied
  #         calling fopen(/etc/ssl/openssl.cnf, rb)
  #
  # /etc/ssl/cert.pem is a SYMLINK to certs/ca-certificates.crt. AppArmor
  # mediates the RESOLVED path, so permitting the link alone would not have been
  # enough; the target is covered by the certs rules below. OpenSSL additionally
  # probes hash-form aliases (/etc/ssl/certs/<8hex>.0) while building a chain —
  # observed in the trace — so the directory and its contents are readable.
  #
  # Read only. Nothing here is writable, and /etc/ssl/private is denied outright:
  # this app presents no client certificate and terminates no TLS, so it has no
  # business reading a system private key.
  /etc/ssl/openssl.cnf   r,
  /etc/ssl/cert.pem      r,
  /etc/ssl/certs/        r,
  /etc/ssl/certs/**      r,
  deny /etc/ssl/private/** rwklx,
  deny /etc/ssl/**       w,

  # ── Name resolution ────────────────────────────────────────────────────────
  # Without these the next failure after TLS is DNS. Enumerated from the same
  # trace rather than discovered one denial at a time.
  /etc/resolv.conf       r,
  /etc/hosts             r,
  /etc/nsswitch.conf     r,
  /etc/netsvc.conf       r,
  /etc/svc.conf          r,
  /etc/services          r,

  # ── musl loader, locale and time data ──────────────────────────────────────
  /etc/                  r,
  /etc/ld-musl-*.path    r,
  /etc/localtime         r,
  /usr/share/zoneinfo/   r,
  /usr/share/zoneinfo/** r,
  /usr/share/icu/**      r,

  # Node's own shared libraries — libcrypto, libssl, libicu, libcares, brotli,
  # nghttp2, libstdc++. `m` as well as `r`: they are mapped executable, and `r`
  # alone fails the mmap.
  #
  # Some of these, and the zoneinfo rules above, are probably also granted by
  # abstractions/base. They are named anyway: `abstractions/base` is the HOST's
  # file, its contents differ between AppArmor versions, and Home Assistant OS's
  # copy cannot be inspected from here. Nothing the runtime provably needs is
  # left depending on a file this repository does not control. The cost is a few
  # redundant read rules; the alternative cost is another dead release.
  /lib/**                mr,
  /usr/lib/**            mr,
  /usr/local/lib/**      mr,

  # ── The application ────────────────────────────────────────────────────────
  /app/                 r,
  /app/**               r,
  /app/run.sh           rix,

  # The ONLY writable location: its own private volume, holding the device key.
  /data/** rwk,
  /tmp/** rwk,

  @{PROC}/@{pid}/** r,

  # Explicitly denied, so a regression is refused rather than merely absent.
  deny /usr/bin/chromium* rwklx,
  deny /var/run/docker.sock rwklx,
  deny /run/docker.sock rwklx,
  deny /usr/local/bin/frpc rwklx,
  deny /config/** rwklx,
  deny /share/** rwklx,
  deny /usr/bin/ssh* rwklx,
}
